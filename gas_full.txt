// 採寸平面図メーカー ⇔ Googleスプレッドシート連携
// このスプレッドシートの「拡張機能」→「Apps Script」に、この中身を全部貼り付けて保存し、
// 「デプロイ」→「新しいデプロイ」→ 種類「ウェブアプリ」→ 実行ユーザー「自分」・アクセスできるユーザー「全員」でデプロイ。
// 表示された「ウェブアプリのURL」を、図面メーカーの「送る・書き出し」→「スプレッドシート連携」に貼り付けます。
//
// ・1図面 = 1行。同じ図面を直すと、その行が書き換わります（右端の「ID」で見分けます）。
// ・「賃料（月額・円）」はスプレッドシートに直接入れてもOK。アプリ側が空欄なら、入れた賃料は消えません。
// ・「坪単価（円/坪）」は 賃料 ÷ 坪 で自動計算されます。
// ・図面そのもの（JSON）は「図面データ」シートに保存されます。アプリが壊れてもここから戻せます。

const LIST = '図面一覧';
const DATA = '図面データ';
const DELS = '削除済み';
const SUM = '集計';
const NAMES = '名前の登録';
// 「名前の登録」シート：メールアドレス → 表示する名前（漢字など）。登録した名前が「測った人」に出ます
function names_(ss) {
  let sh = ss.getSheetByName(NAMES);
  if (!sh) { sh = ss.insertSheet(NAMES); sh.getRange(1, 1, 1, 2).setValues([['メールアドレス', '名前（漢字など）']]).setFontWeight('bold'); sh.setFrozenRows(1); sh.setColumnWidth(1, 260); sh.setColumnWidth(2, 160); }
  const m = {}; if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues().forEach(r => { if (r[0] && r[1]) m[String(r[0]).trim().toLowerCase()] = String(r[1]).trim(); });
  return m;
}

// 「図面一覧」から行を消した図面は「削除済み」に入れる（図面メーカーからも消えます）。
// 図面の中身は「図面データ」に控えとして残すので、「削除済み」シートからIDの行を消せば戻せます。
function deleted_(ss) {
  const dsh = ss.getSheetByName(DELS);
  return dsh && dsh.getLastRow() ? dsh.getRange(1, 1, dsh.getLastRow(), 1).getValues().map(r => String(r[0])).filter(String) : [];
}
function markDeleted_(ss, ids) {
  if (!ids.length) return;
  const dsh = ss.getSheetByName(DELS) || ss.insertSheet(DELS);
  const have = deleted_(ss);
  ids.filter(id => have.indexOf(id) < 0).forEach(id => dsh.appendRow([id, new Date()]));
  try { dsh.hideSheet(); } catch (err) {}
}
function syncRemovedRows_(ss) {
  const list = ss.getSheetByName(LIST), data = ss.getSheetByName(DATA);
  if (!list || !data || data.getLastRow() < 2 || list.getLastRow() < 1) return;
  const lv = list.getDataRange().getValues(); const oid = lv[0].map(String).indexOf('ID'); if (oid < 0) return;
  const inList = {}; for (let i = 1; i < lv.length; i++) inList[String(lv[i][oid])] = 1;
  if (lv.length < 2) return; // 一覧が空っぽ（作り直し中など）のときは何もしない
  const dels = deleted_(ss);
  const gone = data.getRange(2, 1, data.getLastRow() - 1, 1).getValues().map(r => String(r[0])).filter(id => id && !inList[id] && dels.indexOf(id) < 0);
  markDeleted_(ss, gone);
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    syncRemovedRows_(ss);
    let list = ss.getSheetByName(LIST);
    if (!list) { list = ss.getSheets()[0]; list.setName(LIST); }

    // いまの表を読み込んで、新しい列の並びに置き直す（IDのない行＝最初の仮データは捨てる）
    const old = list.getDataRange().getValues();
    const oh = old.length ? old[0].map(String) : [];
    // 表に自分で足した列も残す（「ID」の手前に並べる）
    const head = body.head.concat(['測った人', '図面を見る']);
    oh.forEach(h => { if (h && h !== 'ID' && head.indexOf(h) < 0) head.push(h); });
    head.push('ID');
    const WHO = head.indexOf('測った人'), LINK = head.indexOf('図面を見る');
    const RENT = head.indexOf('賃料（月額・円）');
    const UNIT = head.indexOf('坪単価（円/坪）');
    const TSUBO = head.indexOf('坪');
    const DATE = head.indexOf('採寸日');
    const IDC = head.length - 1;
    const APPN = body.head.length; // ここまでが図面メーカーから来る列
    const oid = oh.indexOf('ID');
    const rows = [];
    for (let i = 1; i < old.length; i++) {
      const id = oid >= 0 ? String(old[i][oid] || '') : '';
      if (!id) continue;
      rows.push(head.map(h => { const k = oh.indexOf(h); return k >= 0 ? old[i][k] : ''; }));
    }
    const idx = {};
    rows.forEach((r, i) => { idx[String(r[IDC])] = i; });

    // どこかの端末で削除された図面は、一覧から外して「削除済み」に記録（他の端末からも消えます）
    markDeleted_(ss, (body.del || []).map(String));
    const delIds = deleted_(ss);
    if (delIds.length) {
      for (let i = rows.length - 1; i >= 0; i--) if (delIds.indexOf(String(rows[i][IDC])) >= 0) rows.splice(i, 1);
      Object.keys(idx).forEach(k => delete idx[k]);
      rows.forEach((r, i) => { idx[String(r[IDC])] = i; });
    }
    // 他の端末で、もっと新しく直された図面は上書きしない
    const dsh0 = ss.getSheetByName(DATA), stale = {};
    if (dsh0 && dsh0.getLastRow() > 1) {
      const mv = {}; dsh0.getRange(2, 1, dsh0.getLastRow() - 1, 5).getValues().forEach(r => { mv[String(r[0])] = Number(r[4]) || 0; });
      body.rows.forEach(x => { let mt = 0; try { mt = Number(JSON.parse(x.json).mt) || 0; } catch (err) {} if ((mv[x.id] || 0) > mt) stale[x.id] = 1; });
    }
    body.rows = body.rows.filter(x => !stale[x.id]);
    body.rows.forEach(x => {
      if (delIds.indexOf(String(x.id)) >= 0) return;
      const nr = x.values.slice(0, APPN).concat(head.slice(APPN, IDC).map(() => ''), [x.id]); nr[WHO] = body.who || '';
      const i = idx[x.id];
      if (i != null) {
        if (rows[i][WHO]) nr[WHO] = rows[i][WHO]; // 最初に測った人を残す
        // 表で直接入れた内容は消さない：図面メーカー側が空欄の項目は、表の値をそのまま残す（坪単価・図面を見るは毎回作り直す）
        for (let k = 0; k < IDC; k++) { if (k === UNIT || k === LINK) continue; if ((nr[k] === '' || nr[k] == null) && rows[i][k] !== '' && rows[i][k] != null) nr[k] = rows[i][k]; }
        rows[i] = nr;
      } else {
        idx[x.id] = rows.length;
        rows.push(nr);
      }
    });

    // 数字の列は数値として入れる（坪単価の計算に使うため）
    const NUMS = ['面積㎡（柱除く）', '坪', '賃料（月額・円）', '精度m'].map(h => head.indexOf(h)).filter(i => i >= 0);
    rows.forEach(r => NUMS.forEach(i => { const v = String(r[i]).replace(/,/g, ''); if (v !== '' && !isNaN(v)) r[i] = Number(v); }));
    rows.sort((a, b) => String(b[DATE]).localeCompare(String(a[DATE])));
    // 測った人：メールアドレスを登録した名前に置きかえる
    try { const nm = names_(ss); rows.forEach(r => { const k = String(r[WHO] || '').trim().toLowerCase(); if (nm[k]) r[WHO] = nm[k]; }); } catch (err) {}
    // 「図面を見る」：押すと図面メーカーでその図面が開く
    let appUrl = 'https://script.google.com/a/macros/uk-corp.co.jp/s/AKfycby0Zyzh7BG8gQ0LBOpXHbVtY-Te-DCNlTMcgsJCdF2rAHxH3V6jgQ9_vbq5TN5wphY/exec';
    try { const u = ScriptApp.getService().getUrl(); if (u && /\/exec$/.test(u)) appUrl = u; } catch (err) {}
    rows.forEach(r => { r[LINK] = '=HYPERLINK("' + appUrl + '?open=' + encodeURIComponent(String(r[IDC])) + '","図面を見る")'; });

    list.clearContents();
    list.getRange(1, 1, 1, list.getMaxColumns()).clearContent();
    list.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold');
    list.setFrozenRows(1);
    if (rows.length) {
      list.getRange(2, 1, rows.length, head.length).setValues(rows);
      // 坪単価 = 賃料 ÷ 坪（どちらかが空なら空欄）
      const rc = colName(RENT + 1), tc = colName(TSUBO + 1);
      const f = rows.map((_, i) => [`=IF(AND(N(${rc}${i + 2})>0,N(${tc}${i + 2})>0),ROUND(${rc}${i + 2}/${tc}${i + 2}),"")`]);
      list.getRange(2, UNIT + 1, rows.length, 1).setFormulas(f).setNumberFormat('#,##0');
      list.getRange(2, RENT + 1, rows.length, 1).setNumberFormat('#,##0');
    }
    // 見やすく：1行1図面で折り返さない・列幅をそろえる・見出しに色・IDは隠す
    try {
      list.getRange(1, 1, Math.max(rows.length + 1, 2), head.length).setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP).setVerticalAlignment('middle');
      list.getRange(1, 1, 1, head.length).setBackground('#e8eef7');
      list.setColumnWidths(1, head.length, 100);
      const W = { '測った人': 190, '図面を見る': 90, '図面名': 170, '物件名': 150, '住所（入力）': 200, '測った場所（GPS）': 220, '地図': 90, '注記': 260, '保存元': 120 };
      head.forEach((h, i) => { if (W[h]) list.setColumnWidth(i + 1, W[h]); });
      list.hideColumns(IDC + 1);
      list.setFrozenColumns(1);
    } catch (err) {}

    // 集計：日ごと・人ごとの採寸件数（今日の件数は一番上）
    try { writeSummary_(ss, rows, DATE, WHO, head.indexOf('作成端末')); } catch (err) {}
    try { ensureSearch_(ss); } catch (err) {}

    // 図面そのもの
    const data = ss.getSheetByName(DATA) || ss.insertSheet(DATA);
    if (data.getLastRow() === 0) {
      data.getRange(1, 1, 1, 6).setValues([['ID', '図面名', '最終更新', '図面データ(JSON)', '更新時刻', '採寸日']]).setFontWeight('bold');
      data.setFrozenRows(1);
    } else if (data.getRange(1, 5).getValue() === '') {
      data.getRange(1, 5, 1, 2).setValues([['更新時刻', '採寸日']]).setFontWeight('bold');
    }
    data.hideColumns(4); // 図面データ(JSON)の列は見なくて大丈夫なので隠す（図面メーカーから開けます）
    try { list.activate(); data.hideSheet(); } catch (err) {} // 図面データのシートごと隠す（中身は図面メーカーが使います）
    data.getRange('F:F').setNumberFormat('@'); // 採寸日は文字のまま（日付に変えられないように）
    const m = data.getLastRow();
    let dids = m > 1 ? data.getRange(2, 1, m - 1, 1).getValues().map(r => String(r[0])) : [];
    body.rows.forEach(x => {
      if (!x.json || x.json.length >= 49000 || delIds.indexOf(String(x.id)) >= 0) return;
      let mt = 0, ed = '';
      try { const o = JSON.parse(x.json); mt = Number(o.mt) || 0; ed = String(o.ed || ''); } catch (err) {}
      const j = dids.indexOf(x.id);
      if (j >= 0) {
        const old = Number(data.getRange(j + 2, 5).getValue()) || 0;
        if (old > mt) return; // 他の端末の、もっと新しい直しを上書きしない
        data.getRange(j + 2, 1, 1, 6).setValues([[x.id, x.values[0], new Date(), x.json, mt, ed]]);
      } else { data.appendRow([x.id, x.values[0], new Date(), x.json, mt, ed]); dids.push(x.id); }
    });
  } finally {
    lock.releaseLock();
  }
  return ContentService.createTextOutput('ok');
}

// 「物件さがし」シート：図面一覧を、賃料・坪単価・面積の順に並べたり、エリアで絞ったりして見るためのシート
// （図面一覧は同期のたびに書き直されるので、並び替えはこのシートでする。ここで何を変えても図面には影響しない）
const SEARCH = '物件さがし';
function makeSearchSheet() { ensureSearch_(SpreadsheetApp.getActiveSpreadsheet()); }
function ensureSearch_(ss) {
  if (ss.getSheetByName(SEARCH)) return;
  const sh = ss.insertSheet(SEARCH);
  const url = ScriptApp.getService().getUrl();
  sh.getRange('A1').setValue('物件さがし（図面一覧から自動で並びます。黄色の欄を変えると、並び順・絞り込み・判定が変わります）').setFontWeight('bold').setFontSize(13);
  sh.getRange('A3:B8').setValues([['並び順', '坪単価が安い順'], ['住所・物件名・図面名に含む文字', ''], ['', ''], ['出店基準：坪数の上限（空欄なら見ない）', 100], ['「惜しい」とする範囲（基準を超えた割合 ％まで）', 15], ['判定で絞る', 'すべて']]);
  sh.getRange('A5').setValue('（例：新北、中山、淡水）').setFontColor('#888888');
  sh.getRange('D3:E9').setValues([['エリア（住所に含む文字）', '坪単価の上限'], ['台北', 1200], ['新北', 800], ['', ''], ['', ''], ['', ''], ['上の行に当てはまらないとき', 3000]]);
  sh.getRange('A3:A8').setFontWeight('bold'); sh.getRange('D3:E3').setFontWeight('bold');
  sh.getRange('B3').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['坪単価が安い順', '賃料が安い順', '面積が広い順', '面積が狭い順', '新しい順'], true).build());
  sh.getRange('B8').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['すべて', '○と△だけ', '△だけ'], true).build());
  for (const r of ['B3', 'B4', 'B6', 'B7', 'B8', 'D4:E8', 'E9']) sh.getRange(r).setBackground('#fff7d6');
  const head = ['判定', '図面名', '物件名', '住所', '階', '坪', '賃料（月額）', '坪単価', '基準の坪単価', '採寸日', '測った人', '図面'];
  sh.getRange(10, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#e8eef7');
  const F = '=LET(h,\'図面一覧\'!A1:AZ1,d,\'図面一覧\'!A2:AZ,c,LAMBDA(n,INDEX(d,0,XMATCH(n,h))),'
    + 'id,c("ID"),nm,c("図面名"),pr,c("物件名"),ad,c("住所（入力）"),fl,c("階"),tb,c("坪"),rent,c("賃料（月額・円）"),up,c("坪単価（円/坪）"),dt,c("採寸日"),who,c("測った人"),'
    + 'lim,MAP(ad,LAMBDA(a,IFERROR(INDEX(FILTER($E$4:$E$8,($D$4:$D$8<>"")*ISNUMBER(SEARCH($D$4:$D$8,a))),1),$E$9))),'
    + 'pct,N($B$7)/100,tl,N($B$6),'
    + 'okU,(up<>"")*(up<=lim),okT,((tl=0)+(tb<=tl))>0,nearU,(up<>"")*(up<=lim*(1+pct)),nearT,((tl=0)+(tb<=tl*(1+pct)))>0,'
    + 'j,IF(up="","賃料なし",IF(okU*okT,"○ 基準内",IF(nearU*nearT,"△ 惜しい","× 基準外"))),'
    + 'q,TRIM($B$4),keep,(id<>"")*IF(q="",1,ISNUMBER(SEARCH(q,nm&" "&pr&" "&ad))),'
    + 'f2,$B$8,keep2,keep*IF(f2="○と△だけ",((LEFT(j,1)="○")+(LEFT(j,1)="△"))>0,IF(f2="△だけ",LEFT(j,1)="△",1)),'
    + 'lk,IF(id="","",HYPERLINK("' + url + '?open="&id,"図面を見る")),'
    + 'M,FILTER(HSTACK(j,nm,pr,ad,fl,tb,rent,up,lim,dt,who,lk),keep2),'
    + 's,$B$3,k,SWITCH(s,"賃料が安い順",7,"面積が広い順",6,"面積が狭い順",6,"新しい順",10,8),o,SWITCH(s,"面積が広い順",FALSE,"新しい順",FALSE,TRUE),'
    + 'IFERROR(SORT(M,k,o),"該当する図面はありません"))';
  sh.getRange('A11').setFormula(F);
  sh.getRange('F11:F').setNumberFormat('0.00'); sh.getRange('G11:I').setNumberFormat('#,##0');
  const R = sh.getRange('A11:L');
  sh.setConditionalFormatRules([
    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=LEFT($A11,1)="△"').setBackground('#fff1c2').setRanges([R]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=LEFT($A11,1)="○"').setBackground('#e3f0dc').setRanges([R]).build()]);
  sh.setFrozenRows(10);
  const W = [90, 170, 150, 220, 50, 70, 100, 90, 100, 100, 110, 90]; W.forEach((w, i) => sh.setColumnWidth(i + 1, w));
  sh.setColumnWidth(1, 260); sh.setColumnWidth(4, 220);
}

function writeSummary_(ss, rows, DATE, WHO, DEV) {
  const sh = ss.getSheetByName(SUM) || ss.insertSheet(SUM);
  const tz = Session.getScriptTimeZone(), today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  const day = v => v instanceof Date ? Utilities.formatDate(v, tz, 'yyyy-MM-dd') : String(v || '').slice(0, 10);
  const who = r => String(r[WHO] || '').split('@')[0] || String(r[DEV] || '') || '（不明）';
  const people = [], cnt = {};
  rows.forEach(r => { const d = day(r[DATE]); if (!d) return; const w = who(r); if (people.indexOf(w) < 0) people.push(w); cnt[d] = cnt[d] || {}; cnt[d][w] = (cnt[d][w] || 0) + 1; });
  const days = Object.keys(cnt).sort().reverse().slice(0, 62);
  const todayN = cnt[today] ? Object.keys(cnt[today]).reduce((a, k) => a + cnt[today][k], 0) : 0;
  const out = [['今日の採寸（' + Number(today.slice(5, 7)) + '/' + Number(today.slice(8, 10)) + '）', todayN + '件'].concat(people.map(() => '')), ['', ''].concat(people.map(() => '')), ['採寸日', '合計'].concat(people)];
  days.forEach(d => { const c = cnt[d]; out.push([d, Object.keys(c).reduce((a, k) => a + c[k], 0)].concat(people.map(w => c[w] || ''))); });
  sh.clearContents();
  sh.getRange(1, 1, out.length, out[0].length).setValues(out);
  sh.getRange(1, 1, 1, 2).setFontWeight('bold').setFontSize(13);
  sh.getRange(3, 1, 1, out[0].length).setFontWeight('bold').setBackground('#e8eef7');
  sh.setFrozenRows(3); sh.setColumnWidth(1, 210); sh.setColumnWidth(2, 80);
  if (out[0].length > 2) sh.setColumnWidths(3, out[0].length - 2, 110);
  sh.getRange(1, 1, out.length, out[0].length).setVerticalAlignment('middle'); sh.getRange(1, 2, out.length, out[0].length - 1).setHorizontalAlignment('center');
}

function colName(n) { let s = ''; while (n > 0) { const m = (n - 1) % 26; s = String.fromCharCode(65 + m) + s; n = Math.floor((n - 1) / 26); } return s; }

// 図面メーカーの「スプレッドシートの図面を開く」から呼ばれます（list=1 で一覧、id=… でその図面）
function doGet(e) {
  const p = (e && e.parameter) || {};
  const data = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(DATA);
  const out = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (p.list) { const lock = LockService.getScriptLock(); if (lock.tryLock(10000)) { try { syncRemovedRows_(ss); } finally { lock.releaseLock(); } } }
  const del = p.list ? deleted_(ss) : [];
  if (p.list == '2') {
    // 図面メーカーの自動同期用：全図面の更新時刻と、削除された図面のID
    if (!data || data.getLastRow() < 2) return out({ rows: [], del: del });
    const v = data.getRange(2, 1, data.getLastRow() - 1, 6).getValues().filter(r => del.indexOf(String(r[0])) < 0);
    // meta=1（過去の図面の一覧用）：場所・物件名・住所・担当者・測った人・自分の図面か
    let meta = null, me = '', myName = '';
    if (p.meta) {
      meta = {};
      const L = ss.getSheetByName(LIST), lv = L ? L.getDataRange().getValues() : [[]], lh = lv[0].map(String);
      const ci = h => lh.indexOf(h), I = ci('ID'), W = ci('測った人'), PR = ci('物件名'), AD = ci('住所（入力）'), ST = ci('担当者');
      for (let i = 1; i < lv.length; i++) { const r = lv[i]; if (I >= 0 && r[I]) meta[String(r[I])] = { who: W >= 0 ? String(r[W] || '') : '', prop: PR >= 0 ? String(r[PR] || '') : '', addr: AD >= 0 ? String(r[AD] || '') : '', staff: ST >= 0 ? String(r[ST] || '') : '' }; }
      try { me = Session.getActiveUser().getEmail() || ''; } catch (err) {}
      try { myName = names_(ss)[me.toLowerCase()] || ''; } catch (err) {}
    }
    return out({ del: del, me: me, rows: v.map(r => {
      let mt = Number(r[4]) || 0, ed = r[5] instanceof Date ? Utilities.formatDate(r[5], Session.getScriptTimeZone(), 'yyyy-MM-dd') : String(r[5] || '');
      let o = null;
      if ((!mt && !ed) || meta) { try { o = JSON.parse(String(r[3])); } catch (err) {} }
      if (!mt && !ed && o) { mt = Number(o.mt) || 0; ed = String(o.ed || ''); }
      const row = { id: String(r[0]), title: String(r[1]), updated: r[2] instanceof Date ? r[2].toISOString() : String(r[2]), mt: mt, ed: ed };
      if (meta) {
        const m = meta[row.id] || {}, I = (o && o.info) || {}, g = o && o.geo;
